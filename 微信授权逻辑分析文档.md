## 1. 概述

### 1.1 技术架构

项目采用多端适配架构，根据运行环境自动选择对应的微信授权方式：

| 运行环境 | 平台标识 | 社交类型(socialType) | 授权方式 |
|---------|---------|---------------------|---------|
| 微信小程序 | `WechatMiniProgram` | 34 | wx.login + 手机号快速验证 |
| 微信公众号(H5) | `WechatOfficialAccount` | 31 | OAuth2.0 网页授权 |
| App | `App` | 32 | 微信开放平台授权 |

### 1.2 核心设计理念

- **统一入口**：通过 `sheep.$platform.useProvider()` 统一调用不同平台的授权方法
- **条件编译**：使用 uni-app 的条件编译 `#ifdef` 实现多端代码隔离
- **Token 管理**：登录成功后自动存储 accessToken 和 refreshToken

## 2. 支持的微信授权场景

### 2.1 登录场景

| 场景 | 描述 | 触发方式 |
|-----|------|---------|
| 静默登录 | 使用 wx.login 获取 code 进行登录 | 自动触发 |
| 手机号快捷登录 | 小程序获取手机号一键登录 | 点击"快捷登录"按钮 |
| 第三方登录 | 点击微信图标进行授权登录 | 点击微信登录按钮 |

### 2.2 绑定场景

| 场景 | 描述 | 触发位置 |
|-----|------|---------|
| 账号绑定微信 | 已登录用户绑定微信账号 | 个人中心 -> 用户信息 |
| 解绑微信 | 已绑定用户解除微信绑定 | 个人中心 -> 用户信息 |

// 社交类型 - 微信小程序
const socialType = 34;

// 静默登录
const login = async () => {
  // 1. 获得微信 code
  const codeResult = await uni.login();
  if (codeResult.errMsg !== 'login:ok') {
    return false;
  }
  // 2. 社交登录
  const loginResult = await AuthUtil.socialLogin(socialType, codeResult.code, 'default');
  if (loginResult.code === 0) {
    setOpenid(loginResult.data.openid);
    return true;
  }
  return false;
};

const mobileLogin = async (e) => {
  if (e.errMsg !== 'getPhoneNumber:ok') {
    return false;

  // 2. 一键登录
  const loginResult = await AuthUtil.weixinMiniAppLogin(e.code, codeResult.code, 'default');
  if (loginResult.code === 0) {
    setOpenid(loginResult.data.openid);
    return true;
  }
  return false;

### 5.2 关键代码 - officialAccount.js

```javascript
// 社交类型 - 微信公众号
const socialType = 31;

// 微信公众号登陆
async function login(code = '', state = '') {
  // 情况一：没有 code 时，去获取 code
  if (!code) {
    const loginUrl = await getLoginUrl();
    if (loginUrl) {
      uni.setStorageSync('returnUrl', location.href);
      window.location = loginUrl;  // 重定向到微信授权页
    }
  // 情况二：有 code 时，使用 code 去自动登录
  } else {
    const loginResult = await AuthUtil.socialLogin(socialType, code, state);
    if (loginResult.code === 0) {
      setOpenid(loginResult.data.openid);
      return loginResult;
    }
  }
  return false;
}

// 获取公众号登陆地址
async function getLoginUrl(event = 'login') {
  const page = getRootUrl() + 'pages/index/login' + '?event=' + event;
  const { code, data } = await AuthUtil.socialAuthRedirect(socialType, page);
  if (code !== 0) {
    return undefined;
  }
  return data;  // 返回微信授权 URL
}
```

<script setup>
import sheep from '@/sheep';
import { onLoad } from '@dcloudio/uni-app';

onLoad(async (options) => {
  // 解析 URL 参数
  new URLSearchParams(location.search).forEach((value, key) => {
    options[key] = value;
  });
  
  const event = options.event;
  const code = options.code;
  const state = options.state;
  
  if (event === 'login') {
    // 场景一：登录
    await sheep.$platform.useProvider().login(code, state);
  } else if (event === 'bind') {
    // 场景二：绑定
    await sheep.$platform.useProvider().bind(code, state);
  }

  // 跳转回原页面
  let returnUrl = uni.getStorageSync('returnUrl');
  if (returnUrl) {
    uni.removeStorage({key:'returnUrl'});
    location.replace(returnUrl);
  } else {
    uni.switchTab({ url: '/' });
  }
});

### 6.3 后端核心逻辑 - MemberAuthServiceImpl

```java
@Override
@Transactional
public AppAuthLoginRespVO socialLogin(AppAuthSocialLoginReqVO reqVO) {
    // 1. 使用 code 授权码，获取社交用户信息
    SocialUserRespDTO socialUser = socialUserApi.getSocialUserByCode(
        UserTypeEnum.MEMBER.getValue(), 
        reqVO.getType(),    // 社交类型：31公众号/34小程序
        reqVO.getCode(),    // 微信返回的 code
        reqVO.getState()
    );
    
    if (socialUser == null) {
        throw exception(AUTH_SOCIAL_USER_NOT_FOUND);
    }

    MemberUserDO user;
    // 2. 情况一：已绑定用户，直接读取
    if (socialUser.getUserId() != null) {
        user = userService.getUser(socialUser.getUserId());
    // 3. 情况二：未绑定，自动注册 + 绑定
    } else {
        user = userService.createUser(
            socialUser.getNickname(), 
            socialUser.getAvatar(), 
            getClientIP(), 
            getTerminal()
        );
        socialUserApi.bindSocialUser(new SocialUserBindReqDTO(
            user.getId(), 
            getUserType().getValue(),
            reqVO.getType(), 
            reqVO.getCode(), 
            reqVO.getState()
        ));
    }

    // 4. 创建 Token 令牌，记录登录日志
    return createTokenAfterLoginSuccess(user, user.getMobile(), 
        LoginLogTypeEnum.LOGIN_SOCIAL, socialUser.getOpenid());
}

@Override
public AppAuthLoginRespVO weixinMiniAppLogin(AppAuthWeixinMiniAppLoginReqVO reqVO) {
    // 1. 获取手机号信息（通过 phoneCode 解密）
    SocialWxPhoneNumberInfoRespDTO phoneNumberInfo = socialClientApi.getWxMaPhoneNumberInfo(
        UserTypeEnum.MEMBER.getValue(), 
        reqVO.getPhoneCode()
    );

    // 2. 根据手机号获取或创建用户
    MemberUserDO user = userService.createUserIfAbsent(
        phoneNumberInfo.getPurePhoneNumber(),
        getClientIP(), 
        TerminalEnum.WECHAT_MINI_PROGRAM.getTerminal()
    );

    // 3. 绑定社交用户
    String openid = socialUserApi.bindSocialUser(new SocialUserBindReqDTO(
        user.getId(), 
        getUserType().getValue(),
        SocialTypeEnum.WECHAT_MINI_PROGRAM.getType(), 
        reqVO.getLoginCode(), 
        reqVO.getState()
    ));

    // 4. 创建 Token 令牌
    return createTokenAfterLoginSuccess(user, user.getMobile(), 
        LoginLogTypeEnum.LOGIN_SOCIAL, openid);
}
```

// 获取 openid
async function getOpenid(force = false) {
  let openid = uni.getStorageSync('openid');
  if (!openid && force) {
    const info = await getInfo();
    if (info && info.openid) {
      openid = info.openid;
      setOpenid(openid);

  return openid;

### 8.3 Token 刷新机制 - request/index.js

```javascript
// 无感知刷新令牌
let requestList = [];
let isRefreshToken = false;

const refreshToken = async (config) => {
  if (!isRefreshToken) {
    isRefreshToken = true;
    const refreshToken = getRefreshToken();
    
    if (!refreshToken) {
      return handleAuthorized();  // 跳转登录
    }
    
    try {
      const result = await AuthUtil.refreshToken(refreshToken);
      if (result.code === 0) {
        // 刷新成功，回放队列请求
        config.header.Authorization = 'Bearer ' + getAccessToken();
        requestList.forEach((cb) => cb());
        requestList = [];
        return request(config);
      }
    } catch (e) {
      return handleAuthorized();
    } finally {
      requestList = [];
      isRefreshToken = false;
    }
  } else {
    // 正在刷新中，加入队列等待
    return new Promise((resolve) => {
      requestList.push(() => {
        config.header.Authorization = 'Bearer ' + getAccessToken();
        resolve(request(config));
      });
    });
  }
};
```

## 10. 常见问题与解决方案

### 10.1 Token 过期处理

当 Token 过期时（返回 401），系统会自动尝试刷新 Token：

1. 使用 refresh-token 调用刷新接口
2. 刷新成功后，自动重试原请求
3. 刷新失败则跳转登录页

### 10.2 多端登录状态同步

- Token 存储在 `uni.setStorageSync` 中
- 用户状态通过 Pinia Store 管理
- 登录成功后自动加载用户信息、购物车等数据

### 10.3 微信 JSSDK 初始化

公众号环境下需要初始化 JSSDK 才能使用微信支付等功能：

```javascript
// sdk-h5-weixin.js
async init(callback) {
  // 调用后端获取签名
  const { data } = await AuthUtil.createWeixinMpJsapiSignature(url);
  
  // 配置 JSSDK
  jweixin.config({
    appId: data.appId,
    timestamp: data.timestamp,
    nonceStr: data.nonceStr,
    signature: data.signature,
    jsApiList: ['chooseWXPay', 'getLocation', ...],
  });
}
```

## 11. 总结

### 11.1 授权方式对比

| 特性 | 小程序 | 公众号 | App |
|-----|-------|-------|-----|
| 授权方式 | wx.login | OAuth2.0 | 开放平台SDK |
| 获取手机号 | getPhoneNumber | 需用户输入 | 需用户输入 |
| 用户信息 | 需额外授权 | 自动获取 | 自动获取 |
| 静默登录 | 支持 | 不支持 | 不支持 |

### 11.2 核心流程总结

1. **前端**：根据运行环境自动选择授权方式
2. **获取 code**：调用微信 API 获取授权码
3. **后端验证**：使用 code 向微信服务器换取 openid
4. **用户处理**：查询或创建用户，绑定社交账号
5. **返回 Token**：生成 accessToken 和 refreshToken
6. **前端存储**：保存 Token，更新登录状态
